  <script>
    const trainingData = [
      { id: 1, color: '빨강', pattern: '점 있음', length: '김', result: '죽음' },
      { id: 2, color: '빨강', pattern: '점 없음', length: '짧음', result: '생존' },
      { id: 3, color: '노랑', pattern: '점 없음', length: '김', result: '생존' },
      { id: 4, color: '갈색', pattern: '점 있음', length: '짧음', result: '죽음' },
      { id: 5, color: '노랑', pattern: '점 있음', length: '짧음', result: '죽음' },
      { id: 6, color: '갈색', pattern: '점 없음', length: '김', result: '생존' }
    ];

    const testData = {
      id: 101,
      color: '노랑',
      pattern: '점 있음',
      length: '김',
      result: '죽음'
    };

    const steps = [1, 2, 3, 4];
    let currentStep = 1;
    let basket1 = [];
    let basket2 = [];
    let prediction = null;

    function getMushroomEmoji(color) {
      if (color === '빨강') return '🍄';
      if (color === '노랑') return '🌕';
      return '🌰';
    }

    function resultClass(result) {
      return result === '죽음' ? 'danger' : 'safe';
    }

    function renderTrainingList() {
      const container = document.getElementById('training-list');
      container.innerHTML = trainingData.map(item => `
        <div class="mushroom-card">
          <div class="mushroom-emoji">${getMushroomEmoji(item.color)}</div>
          <div class="tag-row">
            <span class="tag">색깔: ${item.color}</span>
            <span class="tag">무늬: ${item.pattern}</span>
            <span class="tag">길이: ${item.length}</span>
          </div>
          <span class="result-badge ${resultClass(item.result)}">원시인 결과: ${item.result}</span>
        </div>
      `).join('');
    }

    function renderSortingList() {
      const container = document.getElementById('sorting-list');
      container.innerHTML = trainingData.map(item => `
        <div class="mushroom-card">
          <div class="mushroom-emoji">${getMushroomEmoji(item.color)}</div>
          <div class="tag-row">
            <span class="tag">${item.color}</span>
            <span class="tag">${item.pattern}</span>
            <span class="tag">${item.length}</span>
          </div>
          <span class="result-badge ${resultClass(item.result)}">결과: ${item.result}</span>
          <div class="actions">
            <button onclick="addToBasket(1, ${item.id})">1번 바구니</button>
            <button class="secondary" onclick="addToBasket(2, ${item.id})">2번 바구니</button>
          </div>
        </div>
      `).join('');
    }

    function renderBasketLists() {
      const basket1El = document.getElementById('basket1-list');
      const basket2El = document.getElementById('basket2-list');
      const basket1Review = document.getElementById('basket1-review');
      const basket2Review = document.getElementById('basket2-review');

      basket1El.innerHTML = basket1.length
        ? basket1.map(item => `
          <div class="small-card">${item.color} / ${item.pattern} / ${item.length} / ${item.result}</div>
        `).join('')
        : '<p>아직 담긴 버섯이 없습니다.</p>';

      basket2El.innerHTML = basket2.length
        ? basket2.map(item => `
          <div class="small-card">${item.color} / ${item.pattern} / ${item.length} / ${item.result}</div>
        `).join('')
        : '<p>아직 담긴 버섯이 없습니다.</p>';

      basket1Review.innerHTML = basket1El.innerHTML;
      basket2Review.innerHTML = basket2El.innerHTML;

      updateFeatureFeedback();
    }

    function addToBasket(basketNumber, id) {
      const item = trainingData.find(m => m.id === id);
      basket1 = basket1.filter(m => m.id !== id);
      basket2 = basket2.filter(m => m.id !== id);

      if (basketNumber === 1) basket1.push(item);
      else basket2.push(item);

      renderBasketLists();
    }

    function countFeature(items, key, value) {
      return items.filter(item => item[key] === value).length;
    }

    function summarizeBasket(items) {
      if (!items.length) return '버섯이 없습니다.';

      const colors = ['빨강', '노랑', '갈색'].map(v => `${v} ${countFeature(items, 'color', v)}개`).join(', ');
      const patterns = ['점 있음', '점 없음'].map(v => `${v} ${countFeature(items, 'pattern', v)}개`).join(', ');
      const lengths = ['김', '짧음'].map(v => `${v} ${countFeature(items, 'length', v)}개`).join(', ');
      const results = ['죽음', '생존'].map(v => `${v} ${countFeature(items, 'result', v)}개`).join(', ');

      return `색깔: ${colors}<br>무늬: ${patterns}<br>길이: ${lengths}<br>결과: ${results}`;
    }

    function updateFeatureFeedback() {
      const box = document.getElementById('feature-feedback');
      box.innerHTML = `
        <strong>1번 바구니 요약</strong><br>${summarizeBasket(basket1)}<br><br>
        <strong>2번 바구니 요약</strong><br>${summarizeBasket(basket2)}
      `;
    }

    function renderTestCard() {
      const container = document.getElementById('test-card');
      container.innerHTML = `
        <div class="mushroom-card" style="max-width: 320px; margin-top: 16px;">
          <div class="mushroom-emoji">${getMushroomEmoji(testData.color)}</div>
          <div class="tag-row">
            <span class="tag">색깔: ${testData.color}</span>
            <span class="tag">무늬: ${testData.pattern}</span>
            <span class="tag">길이: ${testData.length}</span>
          </div>
        </div>
      `;
    }

    function showPredictionResult(choice) {
      prediction = choice;
      const resultBox = document.getElementById('result-box');
      resultBox.classList.remove('hidden');

      const correctChoice = testData.result === '죽음' ? '안 먹는다' : '먹는다';
      const isCorrect = choice === correctChoice;

      resultBox.innerHTML = `
        <strong>${isCorrect ? '정답입니다!' : '다시 생각해 볼까요?'}</strong><br><br>
        여러분의 선택: ${choice}<br>
        실제 결과: 원시인 ${testData.result}<br>
        알맞은 선택: ${correctChoice}<br><br>
        <strong>지도학습 연결</strong><br>
        여러분은 앞에서 본 훈련 데이터를 바탕으로 새로운 버섯을 예측했습니다.
        이것은 인공지능이 학습한 뒤 테스트 데이터를 분류하는 과정과 비슷합니다.
      `;
    }

    function showStep(step) {
      currentStep = step;
      steps.forEach(num => {
        document.getElementById(`step-${num}`).classList.add('hidden');
        document.getElementById(`badge-${num}`).classList.remove('active');
      });
      document.getElementById(`step-${step}`).classList.remove('hidden');
      document.getElementById(`badge-${step}`).classList.add('active');
    }

    function resetAll() {
      basket1 = [];
      basket2 = [];
      prediction = null;
      document.getElementById('basket1-name').value = '';
      document.getElementById('basket2-name').value = '';
      document.getElementById('result-box').classList.add('hidden');
      renderBasketLists();
      showStep(1);
    }

    document.getElementById('go-step-2').addEventListener('click', () => showStep(2));
    document.getElementById('back-step-1').addEventListener('click', () => showStep(1));
    document.getElementById('go-step-3').addEventListener('click', () => showStep(3));
    document.getElementById('back-step-2').addEventListener('click', () => showStep(2));
    document.getElementById('go-step-4').addEventListener('click', () => showStep(4));
    document.getElementById('back-step-3').addEventListener('click', () => showStep(3));
    document.getElementById('predict-eat').addEventListener('click', () => showPredictionResult('먹는다'));
    document.getElementById('predict-noeat').addEventListener('click', () => showPredictionResult('안 먹는다'));
    document.getElementById('reset-btn').addEventListener('click', resetAll);

    renderTrainingList();
    renderSortingList();
    renderBasketLists();
    renderTestCard();
  </script>
